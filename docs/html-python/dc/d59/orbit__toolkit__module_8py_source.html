450 @performance_monitor(logger=None)
451 def propagate_tle(self, tle_line1, tle_line2, time_span_hours=6, step_minutes=0.5,
452 return_vectors=True):
453 """
454 Propagate orbit using TLE data
455
456 Args:
457 tle_line1 (str): First line of TLE
458 tle_line2 (str): Second line of TLE
459 time_span_hours (float): Time span to propagate in hours
460 step_minutes (float): Time step in minutes
461 return_vectors (bool): Whether to return position and velocity vectors
462
463 Returns:
464 dict: Dictionary containing propagation results
465 {'epochs': list, 'positions': list, 'velocities': list (if return_vectors),
466 'orbital_elements': list}
467 """
468 try:
469 start_time = time.time()
470 self.logger.log_propagation_start("TLE", {
471 'time_span_hours': time_span_hours,
472 'step_minutes': step_minutes
473 })
474
475 # Create TLE object
476 tle_obj = sc.TwoLineElement(tle_line1, tle_line2)
477 time_jd = tle_obj.get_jd()
478
479 # Initialize arrays
480 epochs = []
481 positions = []
482 velocities = []
483 orbital_elements = []
484
485 # Propagate orbit
486 for dt in np.arange(0, time_span_hours * 60, step_minutes):
487 # Current epoch
488 current_epoch = time_jd + dt * 60 / (24 * 60 * 60)
489 epochs.append(current_epoch)
490
491 # Get position and velocity in ECEF
492 r_ecef = tle_obj.get_position_at(float(dt))
493 v_ecef = tle_obj.get_velocity_at(float(dt))
494
495 # Convert to ECI, matching the visualization function approach
496 r_eci = sc.frames.to_eci(r_ecef, current_epoch)
497 v_eci = sc.frames.to_eci(v_ecef, current_epoch)
498
499 # Store position and velocity if requested
500 if return_vectors:
501 positions.append(r_eci)
502 velocities.append(v_eci)
503
504 # Calculate orbital elements using ECI vectors
505 oe = sc.orbmath.compute_orbital_elements(r_eci, v_eci)
506
507 # Store orbital elements
508 orbital_elements.append({
509 'a': oe.a, # Semi-major axis (km)
510 'e': oe.ecc, # Eccentricity
511 'i': oe.incl, # Inclination (rad)
512 'raan': oe.Omega, # Right Ascension of Ascending Node (rad)
513 'argp': oe.omega, # Argument of Perigee (rad)
514 'nu': oe.nu, # True Anomaly (rad)
515 'M': oe.m # Mean Anomaly (rad)
516 })
517
518 # Prepare results
519 results = {
520 'epochs': epochs,
521 'orbital_elements': orbital_elements
522 }
523
524 # Add position and velocity vectors if requested
525 if return_vectors:
526 results['positions'] = positions
527 results['velocities'] = velocities
528
529 self.logger.log_propagation_end("TLE", time.time() - start_time, True)
530 return results
531
532 except Exception as e:
533 self.logger.log_error("TLE_propagation", e)
534 print(f"Error in TLE propagation: {e}")
535 return None

654
