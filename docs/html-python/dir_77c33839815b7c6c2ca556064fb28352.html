Basic TLE Visualization

from orbit_toolkit import OrbitVisualizer
# Initialize visualizer
visualizer = OrbitVisualizer()
# ISS TLE data (example)
tle_line1 = "1 25544U 98067A 08264.51782528 -.00002182 00000-0 -11606-4 0 2927"

Simple Propagation Analysis

from orbit_toolkit import OrbitPropagator
import matplotlib.pyplot as plt
propagator = OrbitPropagator()
# Propagate orbit
results = propagator.propagate_tle(
tle_line1, tle_line2,
time_span_hours=24,
step_minutes=30
)
if results:
# Extract orbital elements
elements = results['orbital_elements']
epochs = results['epochs']
# Plot semi-major axis evolution
sma_values = [oe['a'] for oe in elements]
time_hours = [(e - epochs[0]) * 24 for e in epochs]
plt.figure(figsize=(10, 6))
plt.plot(time_hours, sma_values)
plt.xlabel('Time (hours)')
plt.ylabel('Semi-major axis (km)')
plt.title('Orbital Decay Analysis')
plt.grid(True)
plt.show()

RK45 Propagation

from orbit_toolkit import OrbitPropagator
propagator = OrbitPropagator()
# Configure perturbations
perturbations = {
'j2': True, # Earth oblateness
'third_body': True, # Moon/Sun effects
'solar_radiation': True, # Solar radiation pressure
'atmospheric_drag': False # Disable for high orbits
}
# High-precision propagation
results = propagator.propagate_rk45(
tdm_wildcard="data/observations/*.tdm",
delta=100,
delta_error=30,
time_span_hours=24,
step_minutes=10,
perturbations=perturbations
)
if results:
print("RK45 propagation completed")
print(f"Computed {len(results['epochs'])} epochs")
# Analyze eccentricity evolution
ecc_values = [oe['e'] for oe in results['orbital_elements']]
print(f"Eccentricity change: {ecc_values[-1] - ecc_values[0]:.6f}")

Perturbation Analysis

from orbit_toolkit import OrbitPropagator
import numpy as np
import matplotlib.pyplot as plt
propagator = OrbitPropagator()
# Test different perturbation combinations
perturbation_configs = {
'Keplerian': {'j2': False, 'third_body': False, 'solar_radiation': False, 'atmospheric_drag': False},
'J2 Only': {'j2': True, 'third_body': False, 'solar_radiation': False, 'atmospheric_drag': False},
'J2 + Third Body': {'j2': True, 'third_body': True, 'solar_radiation': False, 'atmospheric_drag': False},
'All Forces': {'j2': True, 'third_body': True, 'solar_radiation': True, 'atmospheric_drag': True}
}
results_dict = {}
for config_name, perturbations in perturbation_configs.items():
results = propagator.propagate_rk45(
tdm_wildcard="data/*.tdm",
delta=100, delta_error=30,
time_span_hours=48,
perturbations=perturbations
)
if results:
results_dict[config_name] = results
# Analyze position differences
plt.figure(figsize=(12, 8))
reference = results_dict['Keplerian']
ref_positions = np.array(reference['positions'])
for config_name, results in results_dict.items():
if config_name == 'Keplerian':
continue
positions = np.array(results['positions'])
differences = np.linalg.norm(positions - ref_positions, axis=1)
time_hours = np.arange(len(differences)) * 10 / 60 # 10-minute steps to hours
plt.semilogy(time_hours, differences, label=config_name)
plt.xlabel('Time (hours)')
plt.ylabel('Position Difference from Keplerian (km)')
plt.title('Perturbation Effects on Orbit Propagation')
plt.legend()
plt.grid(True)
plt.show()

Batch Processing

import glob
import json
from orbit_toolkit import OrbitPropagator
from concurrent.futures import ThreadPoolExecutor
import os
def process_satellite_file(tle_file):
"""Process a single TLE file and return results"""
propagator = OrbitPropagator()
try:
with open(tle_file, 'r') as f:
lines = f.readlines()
# Extract TLE lines
tle_line1 = lines[1].strip()
tle_line2 = lines[2].strip()
# Propagate orbit
results = propagator.propagate_tle(
tle_line1, tle_line2,
time_span_hours=24,
step_minutes=60 # Hourly data points
)
if results:
# Calculate orbital period
sma = results['orbital_elements'][0]['a']
period_hours = 2 * np.pi * np.sqrt(sma**3 / 398600.4418) / 3600
return {
'file': os.path.basename(tle_file),
'satellite_number': tle_line1[2:7],
'period_hours': period_hours,
'initial_altitude': sma - 6378.137, # Altitude above Earth
'success': True
}
except Exception as e:
return {
'file': os.path.basename(tle_file),
'error': str(e),
'success': False
}
# Process multiple TLE files in parallel
tle_files = glob.glob("data/tle_catalog/*.tle")
with ThreadPoolExecutor(max_workers=4) as executor:
results = list(executor.map(process_satellite_file, tle_files))
# Save results
successful_results = [r for r in results if r['success']]
failed_results = [r for r in results if not r['success']]
print(f"Processed {len(successful_results)} satellites successfully")
print(f"Failed to process {len(failed_results)} satellites")
# Save to JSON
with open('batch_analysis_results.json', 'w') as f:
json.dump({
'successful': successful_results,
'failed': failed_results,
'summary': {
'total_processed': len(results),
'success_rate': len(successful_results) / len(results) * 100
}
}, f, indent=2)
